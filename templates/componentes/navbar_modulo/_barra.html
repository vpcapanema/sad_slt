{# Macro compartilhado dos menus de bloco. Cada bloco do /restrict/ tem o seu proprio componente nesta pasta, que apenas declara os seus itens e chama este macro. Os links sao renderizados dentro da navbar do cabecalho da pagina, substituindo o menu geral: o primeiro item e o inicio do bloco (/restrict/), seguido pelas paginas do modulo. itens: lista de dicionarios com href - destino do link texto - rotulo exibido modo - opcional; quando presente, o item so fica ativo se o query param "modo" da pagina for igual a este valor ou, para agrupar paginas da mesma funcionalidade num menu suspenso: texto - rotulo do menu itens - lista de links no mesmo formato acima Um link so fica ativo quando o caminho bate e, se o href tiver query string (ex.: ?menu=auditoria), cada parametro dela bate com o da pagina atual. #} {% macro _ativo(item) -%} {%- set caminho = request.url.path if request is defined else "" -%} {%- set modo_atual = request.query_params.get("modo") if request is defined and request.query_params is defined else none -%} {%- set teste = namespace(ok=caminho.endswith(item.href.split("?")[0]) and (item.modo is not defined or item.modo == modo_atual)) -%} {%- if teste.ok and "?" in item.href and request is defined -%} {%- for par in item.href.split("?", 1)[1].split("&") if "=" in par -%} {%- if request.query_params.get(par.split("=", 1)[0]) != par.split("=", 1)[1] -%}{%- set teste.ok = false -%}{%- endif -%} {%- endfor -%} {%- endif -%} {{- "1" if teste.ok else "" -}} {%- endmacro %} {% macro barra(rotulo, itens) %} Início do bloco {% for item in itens %} {% if item.itens is defined %} {% set grupo = namespace(ativo=false) %} {% for filho in item.itens %}{% if _ativo(filho) | trim %}{% set grupo.ativo = true %}{% endif %}{% endfor %}
{% for filho in item.itens %} {{ filho.texto }} {% endfor %}
{% else %} {{ item.texto }} {% endif %} {% endfor %} {% endmacro %}
